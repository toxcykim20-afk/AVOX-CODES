function signUp(){ const name = document.getElementById('regName').value.trim(); const phone = document.getElementById('regPhone').value.trim(); const pass = document.getElementById('regPass').value.trim(); const msg = document.getElementById('regMsg'); if(!name || !phone || !pass){ msg.innerHTML = 'Fill all fields'; alert('Fill all fields'); return; } let users = JSON.parse(localStorage.getItem('vox_users') || '[]'); if(users.find(u => u.phone === phone)){ msg.innerHTML = 'Phone already exists, go to SIGN IN'; alert('This phone already exists, now SIGN IN'); showPage('login'); return; } let newUser = {name, phone, pass, balance:0, profit:0}; users.push(newUser); localStorage.setItem('vox_users', JSON.stringify(users)); localStorage.setItem('vox_current', JSON.stringify(newUser)); msg.innerHTML = 'Account created!'; alert('Account created on THIS device! Now you can use it'); showPage('market'); }
