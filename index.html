function signIn(){ const phone = document.getElementById('loginPhone').value.trim(); const pass = document.getElementById('loginPass').value.trim(); const msg = document.getElementById('loginMsg'); if(!phone || !pass){ msg.innerHTML = 'Enter phone + password'; alert('Enter phone + password'); return; } let users = JSON.parse(localStorage.getItem('vox_users') || '[]'); let found = users.find(u => u.phone === phone && u.pass === pass); if(found){ localStorage.setItem('vox_current', JSON.stringify(found)); msg.innerHTML = 'Success! Loading...'; setTimeout(()=>{ showPage('market'); updateAccount(); }, 500); } else { // check if phone exists but wrong pass let phoneExists = users.find(u => u.phone === phone); if(phoneExists){ msg.innerHTML = 'Wrong password!'; alert('Wrong password! Try again. Phone: ' + phone); } else { msg.innerHTML = 'Account not found on this phone. Please SIGN UP first!'; alert('Account not found on this PHONE. Click SIGN UP to create account on this phone first!'); } } }
