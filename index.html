function deposit(){ let phone = prompt('DEPOSIT\n\nEnter M-PESA Phone Number to pay from:\ne.g 0790806650'); if(!phone) return; if(phone.length < 10){ alert('Enter valid phone 07XXXXXXXX'); return; } let amount = prompt('Enter amount to deposit KSh:\nPay to Till: 123456\nAccount: '+phone); if(!amount) return; amount = parseInt(amount); if(amount < 50){ alert('Minimum deposit is KSh 50'); return; } acc.real += amount; myProfit += amount; updateBal(); alert('✅ Deposit Request Created!\n\nAmount: KSh '+amount+'\nFrom: '+phone+'\nTo Till: 123456\nPlease complete M-PESA payment.\nBalance will reflect after payment.'); // Save deposit request so you can see it as admin let deps = JSON.parse(localStorage.getItem('depositReqs')||'[]'); deps.push({name:users[currentUser].name, phone:phone, amount:amount, userPhone:currentUser, time:new Date().toLocaleString(), status:'Pending'}); localStorage.setItem('depositReqs', JSON.stringify(deps)); }
